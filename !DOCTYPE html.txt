<script>
  // ▼▼ https://script.google.com/macros/s/AKfycbzJ67oquEE41xFw57up7Llq-ZLwOb1lXEmhv7xKV4HzGyPTS3sICF7MVq5iPA_Nc2fLdg/exec (Deploy › Manage deployments › Web app URL, ends with /exec)
  const API_URL = 'PASTE_YOUR_WEB_APP_URL_HERE';
  // ▲▲

  const $ = id => document.getElementById(id);
  let pin = '';
  try { pin = localStorage.getItem('guardPin') || ''; } catch (e) {}
  let scanner = null, busy = false, lastCode = '', lastAt = 0, audio = null, wake = null;

  function api(body) {
    return fetch(API_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'text/plain;charset=utf-8' },   // avoids a CORS pre-check
      body: JSON.stringify(Object.assign({ pin: pin }, body))
    }).then(r => r.json());
  }

  function showPin(msg) { $('pinbox').style.display = 'block'; $('main').style.display = 'none'; $('pinmsg').textContent = msg || ''; }
  function hidePin() { $('pinbox').style.display = 'none'; $('main').style.display = 'block'; }
  $('lock').onclick = () => showPin();
  $('savepin').onclick = () => {
    pin = $('pin').value.trim();
    $('pinmsg').textContent = 'Checking…';
    api({ action: 'ping' }).then(r => {
      if (!r.ok) return showPin('Wrong PIN, try again.');
      try { localStorage.setItem('guardPin', pin); } catch (e) {}
      hidePin();
    }).catch(() => showPin('Cannot reach the sheet. Check internet and the URL.'));
  };
  if (!pin) showPin(); else hidePin();

  function beep(f) {
    try { audio = audio || new AudioContext(); const o = audio.createOscillator(); o.frequency.value = f; o.connect(audio.destination); o.start(); o.stop(audio.currentTime + 0.15); } catch (e) {}
  }
  let hideT;
  function show(cls, big, who, sub, ms) {
    $('result').className = cls; $('big').textContent = big; $('who').textContent = who || ''; $('sub').textContent = sub || '';
    clearTimeout(hideT);
    if (ms) hideT = setTimeout(() => { $('result').className = ''; busy = false; }, ms);
  }
  function handle(r) {
    $('net').style.display = 'none';
    if (!r.ok) { beep(220); if (navigator.vibrate) navigator.vibrate([100, 60, 100]); return show('ERR', r.msg === 'Wrong PIN' ? 'WRONG PIN' : 'NOT FOUND', r.msg, '', 2500); }
    beep(r.dir === 'IN' ? 880 : 660); if (navigator.vibrate) navigator.vibrate(120);
    show(r.dir, (r.dup ? 'ALREADY ' : '') + r.dir, r.name, r.block + ' · Room ' + r.room + (r.time ? ' · ' + r.time : ''), 2200);
  }
  function offline() { $('net').style.display = 'block'; beep(220); show('ERR', 'NOT SAVED', 'No connection', 'Write it down by hand', 3000); }

  function onScan(code) {
    const now = Date.now();
    if (busy || (code === lastCode && now - lastAt < 4000)) return;   // same card held in front of camera
    busy = true; lastCode = code; lastAt = now;
    show('WAIT', '…', 'Saving');
    api({ action: 'scan', code: code }).then(handle).catch(offline);
  }

  $('start').onclick = async () => {
    beep(1);   // unlocks sound on iPhone
    try { if ('wakeLock' in navigator) wake = await navigator.wakeLock.request('screen'); } catch (e) {}
    if (scanner) return;
    scanner = new Html5Qrcode('cam');
    scanner.start({ facingMode: 'environment' }, { fps: 10, qrbox: { width: 240, height: 240 } }, onScan, () => {})
      .then(() => { $('start').textContent = 'Camera on — hold card to camera'; $('start').disabled = true; })
      .catch(err => { scanner = null; show('ERR', 'NO CAMERA', 'Allow camera access', String(err).slice(0, 80), 4000); });
  };
  document.addEventListener('visibilitychange', async () => {
    if (document.visibilityState === 'visible' && wake === null && 'wakeLock' in navigator) { try { wake = await navigator.wakeLock.request('screen'); } catch (e) {} }
  });

  // Manual search for forgotten cards
  let qT;
  $('q').addEventListener('input', e => {
    clearTimeout(qT);
    const q = e.target.value.trim();
    if (q.length < 2) { $('results').innerHTML = ''; return; }
    qT = setTimeout(() => api({ action: 'search', q: q }).then(r => {
      const box = $('results'); box.innerHTML = '';
      (r.list || []).forEach(p => ['IN', 'OUT'].forEach(d => {
        const b = document.createElement('button');
        b.className = d === 'IN' ? 'in' : 'out';
        b.textContent = d + ': ' + p.name + ' (' + p.block + ' ' + p.room + ')';
        b.onclick = () => { busy = true; show('WAIT', '…', 'Saving'); api({ action: 'manual', id: p.id, dir: d }).then(handle).catch(offline); $('q').value = ''; box.innerHTML = ''; };
        box.appendChild(b);
      }));
    }).catch(offline), 350);
  });
</script>
